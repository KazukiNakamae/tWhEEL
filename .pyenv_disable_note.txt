# Local Python environment note
#
# The installation guide uses a Conda environment named `twheel` with
# Python 3.12. The environment name is conventional, not an application
# requirement.
#
# If a pyenv override selects a different interpreter, clear it before
# activating the documented environment:
#
#   unset PYENV_VERSION
#   conda activate twheel
#
# Verify the interpreter before installing or running tWhEEL:
#
#   python --version
#   python -c "import yaml; print(yaml.__version__)"
